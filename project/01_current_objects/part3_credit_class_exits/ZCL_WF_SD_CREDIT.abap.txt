CLASS ZCL_WF_SD_CREDIT IMPLEMENTATION.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_WF_SD_CREDIT->BUILD_WI_USERS
* +-------------------------------------------------------------------------------------------------+
* | [--->] IT_USERS                       TYPE        ZWF_USERS_TY
* | [<-()] RT_WI                          TYPE        ZWF_USERS_WI_TY
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD BUILD_WI_USERS.

    LOOP AT IT_USERS INTO DATA(LS_USER).
      APPEND VALUE #( SEQUENCE  = LS_USER-SEQUENCE
                      USER_DESC = LS_USER-USER_DESC ) TO RT_WI.
    ENDLOOP.

  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_WF_SD_CREDIT->CONSTRUCTOR
* +-------------------------------------------------------------------------------------------------+
* | [--->] IO_FINDER                      TYPE REF TO ZCL_WF_SD_FINDER(optional)
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD CONSTRUCTOR.

    MO_FINDER = COND #( WHEN IO_FINDER IS BOUND
                        THEN IO_FINDER
                        ELSE ZCL_WF_SD_FINDER=>CREATE( ) ).

  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_WF_SD_CREDIT->CONVERT_AMOUNT
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_AMOUNT                      TYPE        UKM_COMM_ACTUAL
* | [--->] IV_FROM                        TYPE        UKM_SGM_CURRENCY
* | [--->] IV_TO                          TYPE        UKM_SGM_CURRENCY
* | [<-()] RV_AMOUNT                      TYPE        UKM_COMM_ACTUAL
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD CONVERT_AMOUNT.

    RV_AMOUNT = IV_AMOUNT.

    IF IV_FROM = IV_TO OR IV_FROM IS INITIAL OR IV_TO IS INITIAL.
      RETURN.
    ENDIF.

    SELECT SINGLE MIN( GDATU )
      FROM TCURR
      WHERE FCURR = @IV_FROM
        AND TCURR = @IV_TO
      INTO @DATA(LV_GDATU).

    IF SY-SUBRC <> 0.
      RETURN.
    ENDIF.

    SELECT SINGLE UKURS
      FROM TCURR
      WHERE FCURR = @IV_FROM
        AND TCURR = @IV_TO
        AND GDATU = @LV_GDATU
      INTO @DATA(LV_RATE).

    IF SY-SUBRC = 0 AND LV_RATE <> 0.
      RV_AMOUNT = IV_AMOUNT * LV_RATE.
    ENDIF.

  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_WF_SD_CREDIT->DEFAULT_SEGMENT
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_KKBER                       TYPE        KKBER
* | [<-()] RV_SEGMENT                     TYPE        UKM_CREDIT_SGMNT
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD DEFAULT_SEGMENT.

    RV_SEGMENT = SWITCH #( IV_KKBER
                   WHEN '1400' THEN 'WFICS'
                   WHEN '1600' THEN 'CPXCS'
                   WHEN '1700' THEN 'WFICS(ETP)'
                   ELSE '1000' ).

  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_WF_SD_CREDIT->GET_APPROVERS
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_WF_TYPE                     TYPE        ZWF_TYPE
* | [--->] IV_BUKRS                       TYPE        BUKRS
* | [--->] IV_CREDIT_GROUP                TYPE        UKM_CRED_GROUP
* | [--->] IT_RANGES                      TYPE        TT_RANGE_CODE
* | [--->] IV_VBELN                       TYPE        VBELN_VA(optional)
* | [<-()] RT_USERS                       TYPE        ZWF_USERS_TY
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD GET_APPROVERS.

    LOOP AT IT_RANGES INTO DATA(LV_RANGE).

      RT_USERS = MO_FINDER->GET_APPROVERS_WF(
        IV_WF_TYPE      = IV_WF_TYPE
        IV_BUKRS        = IV_BUKRS
        IV_CREDIT_GROUP = IV_CREDIT_GROUP
        IV_RANGE_CODE   = LV_RANGE
        IV_VBELN        = IV_VBELN ).

      IF RT_USERS IS NOT INITIAL.
        EXIT.
      ENDIF.

    ENDLOOP.

    SORT RT_USERS ASCENDING BY SEQUENCE.

  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_WF_SD_CREDIT->GET_CREDIT_DATA
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_PARTNER                     TYPE        BU_PARTNER
* | [--->] IV_SEGMENT                     TYPE        UKM_CREDIT_SGMNT
* | [--->] IV_SALESORD_DATA               TYPE        TY_ORDER
* | [<-()] RS_CREDIT                      TYPE        TY_CREDIT
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD GET_CREDIT_DATA.

    DATA LT_TOTALS        TYPE TABLE OF UKM_TOTALS.
    DATA LT_TOTALS_SOURCE TYPE TABLE OF UKM_S_TOTALS_SOURCE.
    DATA LV_LOCAL_CURR    TYPE UKM_SGM_CURRENCY.

    SELECT SINGLE CURRENCY
      FROM UKMCRED_SGM0C
      WHERE CREDIT_SGMNT = @IV_SEGMENT
      INTO @RS_CREDIT-CURRENCY.

    CALL FUNCTION 'UKM_COMMTS_READ'
      EXPORTING
        I_PARTNER            = IV_PARTNER
        I_SEGMENT            = IV_SEGMENT
      TABLES
        ET_UKM_TOTALS        = LT_TOTALS
        ET_UKM_TOTALS_SOURCE = LT_TOTALS_SOURCE.

    LOOP AT LT_TOTALS_SOURCE INTO DATA(LS_SOURCE).
      RS_CREDIT-EXPOSURE = RS_CREDIT-EXPOSURE + LS_SOURCE-AMOUNT.
      LV_LOCAL_CURR      = LS_SOURCE-CURRENCY.
    ENDLOOP.

    RS_CREDIT-EXPOSURE = CONVERT_AMOUNT( IV_AMOUNT = RS_CREDIT-EXPOSURE
                                         IV_FROM   = LV_LOCAL_CURR
                                         IV_TO     = RS_CREDIT-CURRENCY ).

    SELECT SINGLE CREDIT_LIMIT
      FROM UKMBP_CMS_SGM
      WHERE PARTNER      = @IV_PARTNER
        AND CREDIT_SGMNT = @IV_SEGMENT
      INTO @RS_CREDIT-LIMIT.

    IF RS_CREDIT-LIMIT <= 0.
      RETURN.
    ENDIF.
    RS_CREDIT-EXPOSURE    += IV_SALESORD_DATA-NETWR.
    RS_CREDIT-EXCEED_CURR = RS_CREDIT-EXPOSURE - RS_CREDIT-LIMIT.
    RS_CREDIT-EXCEED_PER  = ( RS_CREDIT-EXCEED_CURR / RS_CREDIT-LIMIT ) * 100.

  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_WF_SD_CREDIT->GET_EXCEED_RANGES
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_BUKRS                       TYPE        BUKRS
* | [--->] IS_CREDIT                      TYPE        TY_CREDIT
* | [<-()] RT_RANGES                      TYPE        TT_RANGE_CODE
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD GET_EXCEED_RANGES.

    SELECT RANGE_CODE FROM ZWF_RANGES WHERE TYPE         = @IS_CREDIT-CURRENCY
                                      AND   RANGE_FROM  <= @IS_CREDIT-EXCEED_CURR
                                      AND   RANGE_TO    >= @IS_CREDIT-EXCEED_CURR
                                      AND   COMPANY_CODE = @IV_BUKRS
                                      INTO TABLE @RT_RANGES.
    IF RT_RANGES IS NOT INITIAL.
      RETURN.
    ENDIF.
    SELECT RANGE_CODE FROM ZWF_RANGES WHERE TYPE         = '%'
                                      AND   RANGE_FROM  <= @IS_CREDIT-EXCEED_PER
                                      AND   RANGE_TO    >= @IS_CREDIT-EXCEED_PER
                                      AND   COMPANY_CODE = @IV_BUKRS
                                      INTO TABLE @RT_RANGES.
  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_WF_SD_CREDIT->GET_LAST_EXCEPTION
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_CUSTOMER                    TYPE        KUNAG
* | [<-()] RS_EXCEPTION                   TYPE        TY_EXCEPTION
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD GET_LAST_EXCEPTION.

    SELECT SINGLE MAX( CMFRE )
      FROM VBAK
      WHERE CMGST = 'D'
        AND KNKLI = @IV_CUSTOMER
      INTO @RS_EXCEPTION-LAST_DATE.

    IF RS_EXCEPTION-LAST_DATE IS INITIAL.
      RETURN.
    ENDIF.

    SELECT SINGLE AMTBL
      FROM VBAK
      WHERE CMGST = 'D'
        AND KNKLI = @IV_CUSTOMER
        AND CMFRE = @RS_EXCEPTION-LAST_DATE
      INTO @RS_EXCEPTION-LAST_VALUE.

  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_WF_SD_CREDIT->GET_LIMIT_RANGES
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_BUKRS                       TYPE        BUKRS
* | [--->] IV_CURRENCY                    TYPE        UKM_SGM_CURRENCY
* | [--->] IV_LIMIT                       TYPE        UKM_CREDIT_LIMIT
* | [<-()] RT_RANGES                      TYPE        TT_RANGE_CODE
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD GET_LIMIT_RANGES.

    SELECT RANGE_CODE
      FROM ZWF_RANGES
      WHERE TYPE         = @IV_CURRENCY
        AND RANGE_FROM  <= @IV_LIMIT
        AND RANGE_TO    >= @IV_LIMIT
        AND COMPANY_CODE = @IV_BUKRS
      INTO TABLE @RT_RANGES.

  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_WF_SD_CREDIT->GET_MAX_LEVEL
* +-------------------------------------------------------------------------------------------------+
* | [--->] IT_USERS                       TYPE        ZWF_USERS_TY
* | [<-()] RV_LEVEL                       TYPE        INT2
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD GET_MAX_LEVEL.

    RV_LEVEL = REDUCE INT2( INIT M = 0
                            FOR LS IN IT_USERS
                            NEXT M = COND #( WHEN LS-SEQUENCE > M
                                             THEN LS-SEQUENCE
                                             ELSE M ) ).

  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_WF_SD_CREDIT->GET_MIN_LEVEL
* +-------------------------------------------------------------------------------------------------+
* | [--->] IT_USERS                       TYPE        ZWF_USERS_TY
* | [<-()] RV_LEVEL                       TYPE        INT2
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD GET_MIN_LEVEL.

    IF IT_USERS IS INITIAL.
      RETURN.
    ENDIF.

    RV_LEVEL = IT_USERS[ 1 ]-SEQUENCE.

  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_WF_SD_CREDIT->GET_ORDER_DATA
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_VBELN                       TYPE        VBELN_VA
* | [<-()] RS_ORDER                       TYPE        TY_ORDER
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD GET_ORDER_DATA.

    SELECT SINGLE VBELN,
                  KKBER,
                  BUKRS_VF AS BUKRS,
                  SPART,
                  ERNAM,
                  NETWR,
                  WAERK

      FROM VBAK
      WHERE VBELN = @IV_VBELN
      INTO CORRESPONDING FIELDS OF @RS_ORDER.

    SELECT SINGLE WERKS
      FROM VBAP
      WHERE VBELN  = @IV_VBELN
        AND WERKS <> @SPACE
      INTO @RS_ORDER-WERKS.
  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_WF_SD_CREDIT->GET_OVERDUE_AMOUNT
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_PARTNER                     TYPE        BU_PARTNER
* | [--->] IV_BUKRS                       TYPE        BUKRS
* | [<-()] RS_OVERDUE                     TYPE        TY_OVERDUE
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD GET_OVERDUE_AMOUNT.

    DATA LT_ITEMS TYPE TABLE OF RFPOS.

    SELECT SINGLE ZTERM
      FROM KNB1
      WHERE KUNNR = @IV_PARTNER
        AND BUKRS = @IV_BUKRS
      INTO @DATA(LV_ZTERM).

    IF SY-SUBRC <> 0.
      RETURN.
    ENDIF.

    SELECT SINGLE TEXT1
      FROM T052U
      WHERE ZTERM = @LV_ZTERM
        AND SPRAS = @SY-LANGU
      INTO @RS_OVERDUE-TERM_TEXT.

    SELECT SINGLE ZTAG1
      FROM T052
      WHERE ZTERM = @LV_ZTERM
      INTO @DATA(LV_DAYS).

    RS_OVERDUE-PAYMENT_TERM = LV_DAYS.

    CALL FUNCTION 'CUSTOMER_OPEN_ITEMS'
      EXPORTING
        BUKRS         = IV_BUKRS
        KUNNR         = IV_PARTNER
      TABLES
        T_POSTAB      = LT_ITEMS
      EXCEPTIONS
        NO_OPEN_ITEMS = 1
        OTHERS        = 2.

    IF SY-SUBRC <> 0 OR LT_ITEMS IS INITIAL.
      RETURN.
    ENDIF.

    LOOP AT LT_ITEMS INTO DATA(LS_ITEM)
         WHERE BUDAT <= SY-DATUM - RS_OVERDUE-PAYMENT_TERM.
      RS_OVERDUE-AMOUNT = RS_OVERDUE-AMOUNT + LS_ITEM-DMBE2.
    ENDLOOP.

  ENDMETHOD.

      INTO @RS_OVERDUE-TERM_TEXT WHERE ZTERM = @LV_ZTERM AND TEXT1 <> @SPACE.
    LOOP AT LT_DATA INTO LS_DATA.
      RS_OVERDUE-AMOUNT = RS_OVERDUE-AMOUNT + LS_DATA-OVR_DUE.
    ENDLOOP.
    RS_OVERDUE-DAYS = SY-DATUM - LV_DUE_DATE.
    IF RS_OVERDUE-PAYMENT_TERM <> 0.
      RS_OVERDUE-PERCENT = ( RS_OVERDUE-DAYS / RS_OVERDUE-PAYMENT_TERM ) * 100.
    ENDIF.
  ENDMETHOD.

* | [--->] IV_PARTNER                     TYPE        BU_PARTNER
* | [<-()] RS_PARTNER                     TYPE        TY_PARTNER_BP
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD GET_PARTNER_INFO.

    SELECT SINGLE CONCAT_WITH_SPACE( NAME_ORG1, NAME_ORG2, 1 )
      FROM BUT000
      WHERE PARTNER = @IV_PARTNER
      INTO @RS_PARTNER-BP_NAME.

    IF RS_PARTNER-BP_NAME IS INITIAL.
      SELECT SINGLE CONCAT_WITH_SPACE( NAME_FIRST, NAME_LAST, 1 )
        FROM BUT000
        WHERE PARTNER = @IV_PARTNER
        INTO @RS_PARTNER-BP_NAME.
    ENDIF.

    RS_PARTNER-BP_ID = IV_PARTNER.
    SHIFT RS_PARTNER-BP_ID LEFT DELETING LEADING '0'.

    SELECT SINGLE CREDIT_GROUP
      FROM UKMBP_CMS
      WHERE PARTNER = @IV_PARTNER
      INTO @RS_PARTNER-CREDIT_GROUP.

    SELECT SINGLE BUKRS
      FROM KNB1
      WHERE KUNNR = @IV_PARTNER
      INTO @RS_PARTNER-BUKRS.

  ENDMETHOD.

* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_WF_SD_CREDIT->CLOSE_OPEN_LOGS
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_VBELN                       TYPE        VBELN_VA
* | [--->] IV_REASON                      TYPE        STRING
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD CLOSE_OPEN_LOGS.

    "--- exceed ------------------------------------------------------
    SELECT LOG_ID
      FROM ZWF_LOG_EX_H
      WHERE VBELN  = @IV_VBELN
        AND STATUS = @ZCL_WF_LOG_EX=>GC_STATUS-IN_PROCESS
      INTO TABLE @DATA(LT_EX).

    IF LT_EX IS NOT INITIAL.

      DATA(LO_EX) = NEW ZCL_WF_LOG_EX( ).

      LOOP AT LT_EX INTO DATA(LV_EX).

        LO_EX->ADD_ITEM( IV_LOG_ID = CONV ZWF_LOG_ID( LV_EX )
                         IV_ACTION = ZCL_WF_LOG_EX=>GC_ACTION-KILL
                         IV_NOTE   = CONV #( IV_REASON ) ).

        LO_EX->FINISH( IV_LOG_ID = CONV ZWF_LOG_ID( LV_EX )
                       IV_STATUS = ZCL_WF_LOG_EX=>GC_STATUS-KILLED ).

      ENDLOOP.

    ENDIF.

    "--- overdue -----------------------------------------------------
    SELECT LOG_ID
      FROM ZWF_LOG_OD_H
      WHERE VBELN  = @IV_VBELN
        AND STATUS = @ZCL_WF_LOG_OD=>GC_STATUS-IN_PROCESS
      INTO TABLE @DATA(LT_OD).

    IF LT_OD IS NOT INITIAL.

      DATA(LO_OD) = NEW ZCL_WF_LOG_OD( ).

      LOOP AT LT_OD INTO DATA(LV_OD).

        LO_OD->ADD_ITEM( IV_LOG_ID = CONV ZWF_LOG_ID( LV_OD )
                         IV_ACTION = ZCL_WF_LOG_OD=>GC_ACTION-KILL
                         IV_NOTE   = CONV #( IV_REASON ) ).

        LO_OD->FINISH( IV_LOG_ID = CONV ZWF_LOG_ID( LV_OD )
                       IV_STATUS = ZCL_WF_LOG_OD=>GC_STATUS-KILLED ).

      ENDLOOP.

    ENDIF.

  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_WF_SD_CREDIT->FIND_RUNNING
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_VBELN                       TYPE        VBELN_VA
* | [<-()] RT_WI                          TYPE        TT_WI
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD FIND_RUNNING.

    CHECK IV_VBELN IS NOT INITIAL.

    DATA LV_RC       TYPE SY-SUBRC.
    DATA LT_TASK     TYPE STANDARD TABLE OF SWR_TASK.
    DATA LT_WORKLIST TYPE STANDARD TABLE OF SWR_WIHDR.
    DATA LS_POR      TYPE SIBFLPORB.

    LS_POR = VALUE #( INSTID = IV_VBELN
                      TYPEID = 'BUS2032'
                      CATID  = 'BO' ).

    CALL FUNCTION 'SAP_WAPI_WORKITEMS_TO_OBJECT'
      EXPORTING
        OBJECT_POR               = LS_POR
        OBJTYPE                  = CONV SWOTOBJID-OBJTYPE( 'BUS2032' )
        OBJKEY                   = CONV SWOTOBJID-OBJKEY( IV_VBELN )
        TOP_LEVEL_ITEMS          = CONV SWR_STRUCT-TOP_LEVEL( 'X' )
        SELECTION_STATUS_VARIANT = CONV SWR_STRUCT-SELSTAVAR( 0001 )
        TEXT                     = CONV XFELD( 'X' )
        OUTPUT_ONLY_TOP_LEVEL    = CONV XFELD( SPACE )
        LANGUAGE                 = SY-LANGU
        DETERMINE_TASK_FILTER    = CONV XFELD( 'X' )
        REMOVED_OBJECTS          = CONV XFELD( SPACE )
      IMPORTING
        RETURN_CODE              = LV_RC
      TABLES
        TASK_FILTER              = LT_TASK
        WORKLIST                 = LT_WORKLIST.

    CHECK LV_RC = 0.

    LOOP AT LT_WORKLIST ASSIGNING FIELD-SYMBOL(<FS_WI>) WHERE WI_STAT = 'STARTED'.
      CHECK <FS_WI> IS ASSIGNED.
      SELECT SINGLE TOP_TASK FROM SWWWIHEAD INTO @DATA(TOP_TASK) WHERE WI_ID = @<FS_WI>-WI_ID .
      CHECK TOP_TASK = 'WS99000010'.
      APPEND <FS_WI>-WI_ID TO RT_WI.
    ENDLOOP.
  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_WF_SD_CREDIT->KILL_RUNNING_WORKFLOW
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_VBELN                       TYPE        VBELN_VA
* | [--->] IV_REASON                      TYPE        STRING(optional)
* | [<-()] RV_KILLED                      TYPE        ABAP_BOOL
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD KILL_RUNNING_WORKFLOW.

    CHECK IV_VBELN IS NOT INITIAL.

    DATA(LV_REASON) = COND STRING(
      WHEN IV_REASON IS NOT INITIAL
      THEN IV_REASON
      ELSE |Workflow killed, sales order changed| ).

    DATA(LT_WI) = FIND_RUNNING( IV_VBELN ).

    DATA LV_NEW_STATUS TYPE SWWWIHEAD-WI_STAT.
    DATA LV_CODE       TYPE SWWWIRET-CODE VALUE 0000.

    LOOP AT LT_WI INTO DATA(LV_WI).

      DATA : MESSAGE TYPE SWF_T100MS.
      CALL FUNCTION 'SWW_WI_ADMIN_CANCEL'
        EXPORTING
          WI_ID                       = CONV SWWWIHEAD-WI_ID( LV_WI )
          DO_COMMIT                   = ' '
          LOG_MESSAGE                 = MESSAGE
        IMPORTING
          NEW_STATUS                  = LV_NEW_STATUS
        EXCEPTIONS
          UPDATE_FAILED               = 1
          NO_AUTHORIZATION            = 2
          INFEASIBLE_STATE_TRANSITION = 3
          OTHERS                      = 4.
      IF SY-SUBRC = 0.
        RV_KILLED = ABAP_TRUE.
      ENDIF.
*      CALL FUNCTION 'SWW_WI_ADMIN_COMPLETE'
*        EXPORTING
*          WI_ID                       = CONV SWWWIHEAD-WI_ID( LV_WI )
*          DO_COMMIT                   = CONV SWWCOMMIT-COMMITFLAG( 'X' )
*          AUTHORIZATION_CHECKED       = CONV SWWCOMMIT-AUTHCKFLAG( SPACE )
*          PRECONDITIONS_CHECKED       = CONV SWWCOMMIT-CHECKFLAG( SPACE )
*          RETURN_CODE                 = CONV SWWWIRET-CODE( LV_CODE )
*          EXECUTED_BY_USER            = SY-UNAME
*          EXECUTED_BY_ADDRESS         = CONV SWWADDRKEY( SPACE )
*          CALLED_BY_PROXY             = CONV SWWCOMMIT-COMMITFLAG( SPACE )
*        IMPORTING
*          NEW_STATUS                  = LV_NEW_STATUS
*        EXCEPTIONS
*          UPDATE_FAILED               = 1
*          NO_AUTHORIZATION            = 2
*          INFEASIBLE_STATE_TRANSITION = 3
*          OTHERS                      = 4.
*
*      IF SY-SUBRC = 0.
*        RV_KILLED = ABAP_TRUE.
*      ENDIF.

    ENDLOOP.

    "--- close the logs whether or not a work item was found ---------
    CLOSE_OPEN_LOGS( IV_VBELN  = IV_VBELN
                     IV_REASON = LV_REASON ).

    "--- release the order lock so a new check can run ---------------
    DELETE FROM ZSD_WF_SO WHERE SALES_DOCUMENT = IV_VBELN.

  ENDMETHOD.

* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_WF_SD_CREDIT->FIND_RUNNING_BP
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_PARTNER                     TYPE        BU_PARTNER
* | [<-()] RT_WI                          TYPE        TT_WI
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD FIND_RUNNING_BP.

    CHECK IV_PARTNER IS NOT INITIAL.

    TYPES: BEGIN OF TY_OBJ,
             POR      TYPE SIBFLPORB,
             TOP_TASK TYPE SWWWIHEAD-TOP_TASK,
           END OF TY_OBJ.

    DATA LT_OBJ       TYPE STANDARD TABLE OF TY_OBJ WITH EMPTY KEY.
    DATA LV_RC        TYPE SY-SUBRC.
    DATA LT_TASK      TYPE STANDARD TABLE OF SWR_TASK.
    DATA LT_WORKLIST  TYPE STANDARD TABLE OF SWR_WIHDR.

    LT_OBJ = VALUE #(
      ( POR      = VALUE #( INSTID = IV_PARTNER TYPEID = 'BUS1006'        CATID = 'BO' )
        TOP_TASK = 'WS99000007' )
      ( POR      = VALUE #( INSTID = IV_PARTNER TYPEID = 'CL_UKM_ACCOUNT' CATID = 'CL' )
        TOP_TASK = 'WS99000007' )
    ).

    LOOP AT LT_OBJ INTO DATA(LS_OBJ).

      CLEAR: LT_TASK, LT_WORKLIST, LV_RC.

      CALL FUNCTION 'SAP_WAPI_WORKITEMS_TO_OBJECT'
        EXPORTING
          OBJECT_POR               = LS_OBJ-POR
          OBJTYPE                  = CONV SWOTOBJID-OBJTYPE( LS_OBJ-POR-TYPEID )
          OBJKEY                   = CONV SWOTOBJID-OBJKEY( LS_OBJ-POR-INSTID )
          TOP_LEVEL_ITEMS          = CONV SWR_STRUCT-TOP_LEVEL( 'X' )
          SELECTION_STATUS_VARIANT = CONV SWR_STRUCT-SELSTAVAR( 0001 )
          TEXT                     = CONV XFELD( 'X' )
          OUTPUT_ONLY_TOP_LEVEL    = CONV XFELD( SPACE )
          LANGUAGE                 = SY-LANGU
          DETERMINE_TASK_FILTER    = CONV XFELD( 'X' )
          REMOVED_OBJECTS          = CONV XFELD( SPACE )
        IMPORTING
          RETURN_CODE              = LV_RC
        TABLES
          TASK_FILTER              = LT_TASK
          WORKLIST                 = LT_WORKLIST.

      CHECK LV_RC = 0.

      LOOP AT LT_WORKLIST ASSIGNING FIELD-SYMBOL(<FS_WI>) WHERE WI_STAT = 'STARTED'.
        CHECK <FS_WI> IS ASSIGNED.
        SELECT SINGLE TOP_TASK FROM SWWWIHEAD INTO @DATA(TOP_TASK) WHERE WI_ID = @<FS_WI>-WI_ID.
        CHECK TOP_TASK = LS_OBJ-TOP_TASK.
        APPEND <FS_WI>-WI_ID TO RT_WI.
      ENDLOOP.

    ENDLOOP.

    SORT RT_WI.
    DELETE ADJACENT DUPLICATES FROM RT_WI.

  ENDMETHOD.

  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_WF_SD_CREDIT->IS_START_ALLOWED
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_VBELN                       TYPE        VBELN_VA
* | [<-()] RV_START                       TYPE        ABAP_BOOL
* | [!CX!] ZCX_WF_SD
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD IS_START_ALLOWED.

    DATA LV_ERDAT    TYPE ERDAT.
    DATA LV_START_ON TYPE DATS.
    DATA LV_BLOCK    TYPE CHAR1.

    CLEAR RV_START.

    IF IV_VBELN IS INITIAL.
      RETURN.
    ENDIF.

    SELECT SINGLE ERDAT FROM VBAK
      WHERE VBELN = @IV_VBELN
      INTO @LV_ERDAT.

    IF SY-SUBRC <> 0.
      RAISE EXCEPTION TYPE ZCX_WF_SD
        EXPORTING
          TEXTID   = ZCX_WF_SD=>ORDER_NOT_FOUND
          IV_ATTR1 = IV_VBELN.
    ENDIF.

    LV_START_ON = GET_WF_START_DATE( ).

    IF LV_START_ON IS NOT INITIAL AND LV_ERDAT < LV_START_ON.
      RETURN.
    ENDIF.

    LV_BLOCK = CHECK_SO_BLOCK( IV_VBELN ).

    RV_START = XSDBOOL( LV_BLOCK = 'X' ).

  ENDMETHOD.
